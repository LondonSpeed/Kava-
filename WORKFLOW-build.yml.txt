name: Build APK
on:
  workflow_dispatch:
  push:
    branches: [main, master]
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: npm install
      - run: npx cap add android
      - name: Icons
        run: |
          R=android/app/src/main/res
          find $R -name 'ic_launcher_foreground.png' -exec cp icons/foreground.png {} \;
          find $R -name 'ic_launcher.png' -exec cp icons/icon-192.png {} \;
          find $R -name 'ic_launcher_round.png' -exec cp icons/icon-192.png {} \;
          sed -i 's/#FFFFFF/#3182ce/' $R/values/ic_launcher_background.xml || true
      - run: npx cap sync android
      - run: cd android && chmod +x gradlew && ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: Piibel-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
